//  HARD RULE FOR THIS PROPERTY: nothing here may reference a collection.
//  Named formulas do not receive a collection's schema, so any formula that
//  reads colWindow (or any other collection) reports "name isn't recognized"
//  for every one of its columns, and every formula downstream fails with it.
//  Everything derived from colWindow therefore lives in collections built by
//  btnApplyFilters on scrTimeline, not here.
//
//  App.OnStart is not used and must be left empty.
//
//  Paste this before any screen: every screen references the names below.
// ═══════════════════════════════════════════════════════════════════════════

// ═══════════ Canvas geometry (1366 x 768) ═══════════
LabelW   = 172;    // width of the left category-name column
PlotX    = 180;    // absolute X of the plot area left edge
PlotW    = 1152;   // plot area width
RowMinH  = 56;     // minimum height of a category row
LaneH    = 24;     // height of one marker lane inside a row
TopY     = 198;    // top of the timeline area
PlotH    = 560;    // timeline area height

// Reserved; not referenced by any screen. The marker galleries use
// TemplateSize = LaneH and lay out one lane per marker themselves.
MarkerSlot = 1;

// ═══════════ Time window ═══════════
MonthsVisible = 3;

// The window position is stored as a MONTH OFFSET (a number), not as a date.
// A date variable would be circular: TimelineStart reads it, and the navigation
// arrows set it from TimelineStart. Power Fx cannot infer a type through that
// circle, the variable stays untyped, and every formula reading it fails.
// A number typed by the literal 0 in Coalesce breaks the circle.
TimelineStart = DateAdd(
    Date(Year(Today()), Month(Today()), 1),
    Coalesce(varMonthOffset, 0),
    TimeUnit.Months
);
TimelineEnd   = DateAdd(TimelineStart, MonthsVisible, TimeUnit.Months);
TimelineDays  = DateDiff(TimelineStart, TimelineEnd, TimeUnit.Days);   // 89..92
PxPerDay      = PlotW / TimelineDays;                                  // ~12.6 px per day

// How many records the window ACTUALLY holds — counted server-side, not capped
// by the data row limit. Comparing it with CountRows(colWindow) detects a
// truncated result at any Data row limit setting, including a forgotten 500.
// A hardcoded ">= 2000" check would not.
WindowServerCount =
    CountIf(
        'SEL Engagement Activities',
        'Start Date' <  TimelineEnd,
        'End Date'   >= TimelineStart
    );

//  A description of the active filters, printed in the PDF header — without it an
//  exported timeline cannot be interpreted.
//
//  Reads the shared filter VARIABLES, not the dropdown controls. A formula that
//  reads controls is tied to one screen and breaks when those controls change
//  type; the variables are the filter state itself.
ActiveFilterText =
    With(
        { parts:
            Concat(
                Filter(
                    [ varFProgramme, varFProject, varFStakeholder,
                      varFSite, varFLAA, varFDelivery ],
                    !IsBlank(Value)
                ),
                Value,
                "  ·  "
            )
        },
        If(IsBlank(parts), "No filters applied", parts)
    );
